\begin{mathbox}{Definition: Consciousness Sheaf}

\section{The Second Chern Character}

\subsection{Topological Definition}

\begin{mathbox}{Definition: Second Chern Character}
For a coherent sheaf $\mathcal{F}$ on a variety $\variety$, the second Chern character is:
\begin{equation}
\ch_2(\mathcal{F}) = \frac{1}{2}\left(\ch_1(\mathcal{F})^2 - 2\chern_2(\mathcal{F})\right)
\end{equation}
where $\ch_1$ is the first Chern character and $\chern_2$ is the second Chern class.
\end{mathbox}
